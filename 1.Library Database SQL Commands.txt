-- Create and use the database
CREATE DATABASE IF NOT EXISTS library;
USE library;


-- Create BOOK table
CREATE TABLE BOOK (
  Book_ISBN VARCHAR(20) PRIMARY KEY,
  Title VARCHAR(100) NOT NULL,
  Publisher_Name VARCHAR(50),
  Price DECIMAL(10,2) CHECK (Price > 0),
  Date_Of_Publication DATE,
  Book_Copy INT
);


-- Create BOOK_AUTHORS table
CREATE TABLE BOOK_AUTHORS (
  Book_ISBN VARCHAR(20),
  Author_Name VARCHAR(50),
  Author_City VARCHAR(50),
  PRIMARY KEY (Book_ISBN, Author_Name),
  FOREIGN KEY (Book_ISBN) REFERENCES BOOK(Book_ISBN)
);

-- a) Create view BOOK_AUTHOR_INFO (ascending order by ISBN)
CREATE VIEW BOOK_AUTHOR_INFO AS
SELECT B.Book_ISBN, B.Title, A.Author_Name
FROM BOOK B
JOIN BOOK_AUTHORS A
ON B.Book_ISBN = A.Book_ISBN
ORDER BY B.Book_ISBN ASC;
