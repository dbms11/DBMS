-- 1. Create database and use it
CREATE DATABASE IF NOT EXISTS shopDB;
USE shopDB;

-- 2. Create PRODUCT table
DROP TABLE IF EXISTS PRODUCT;
CREATE TABLE PRODUCT (
    P_ID INT PRIMARY KEY,
    Model VARCHAR(50),
    Price DECIMAL(10,2),
    Name VARCHAR(100),
    Date_of_Manufacture DATE,
    Date_of_Expiry DATE
);

-- 4. (a) Name and date_of_expiry for products with price > 500
SELECT 'Query (a):' AS Description;
SELECT Name, Date_of_Expiry
FROM PRODUCT
WHERE Price > 500;

-- 5. (b) Name, P_ID, and Price for products manufactured after 01-01-2018
SELECT 'Query (b):' AS Description;
SELECT Name, P_ID, Price
FROM PRODUCT
WHERE Date_of_Manufacture > '2018-01-01';

-- 6. (c) Name, Date_of_Manufacture, Date_of_Expiry for products with price between 5000 and 10000
SELECT 'Query (c):' AS Description;
SELECT Name, Date_of_Manufacture, Date_of_Expiry
FROM PRODUCT
WHERE Price BETWEEN 5000 AND 10000;

-- 7. (d) Name, P_ID, Model for products expiring after 2 months from today
SELECT 'Query (d):' AS Description;
SELECT Name, P_ID, Model
FROM PRODUCT
WHERE Date_of_Expiry > DATE_ADD(CURDATE(), INTERVAL 2 MONTH);
