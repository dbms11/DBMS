-- 1. Create database and use it
CREATE DATABASE IF NOT EXISTS libraryDB;
USE libraryDB;

-- 2. Create CATALOG table
DROP TABLE IF EXISTS CATALOG;
CREATE TABLE CATALOG (
    BOOK_ID INT PRIMARY KEY,
    BOOK_TITLE VARCHAR(100),
    AUTHOR VARCHAR(50),
    AUTHOR_ID VARCHAR(20),
    PUBLISHER_ID VARCHAR(50),
    CATEGORY_ID VARCHAR(50),
    YEAR INT,
    ISBN VARCHAR(20),
    PRICE DECIMAL(10,2)
);

-- 4. (a) Display all books of CATEGORY_ID = "COMPUTERS"
SELECT 'Query (a): Books in CATEGORY_ID = COMPUTERS' AS Description;
SELECT * FROM CATALOG

-- 5. (b) List all books whose PRICE >= 1000
SELECT 'Query (b): Books with PRICE >= 1000' AS Description;
SELECT * FROM CATALOG
WHERE PRICE >= 1000;

-- 6. (c) List all books whose PUBLISHER_ID = "Tata McGraw-Hill"

-- 7. (d) List all books whose YEAR = 2013
