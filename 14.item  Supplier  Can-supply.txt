-- Create and use database
CREATE DATABASE IF NOT EXISTS supply_db;
USE supply_db;

-- 1️⃣ Create tables
CREATE TABLE Item (
  item_code INT PRIMARY KEY,
  item_name VARCHAR(100),
  qty_in_stock INT,
  reorder_level INT
);

CREATE TABLE Supplier (
  supplier_code INT PRIMARY KEY,
  supplier_name VARCHAR(100),
  address VARCHAR(100)
);

CREATE TABLE Can_supply (
  supplier_code INT,
  item_code INT,
  PRIMARY KEY (supplier_code, item_code),
  FOREIGN KEY (supplier_code) REFERENCES Supplier(supplier_code),
  FOREIGN KEY (item_code) REFERENCES Item(item_code)
);

-- 4️⃣ List all suppliers from “Madison”
SELECT supplier_name, address
FROM Supplier
WHERE address = 'Madison';

-- 5️⃣ List all suppliers who can supply a given item (example: item_code = 1)
SELECT S.supplier_name, I.item_name
FROM Supplier S
JOIN Can_supply C ON S.supplier_code = C.supplier_code
JOIN Item I ON C.item_code = I.item_code
WHERE I.item_code = 1;

-- 6️⃣ List all items which cannot be supplied by a given company (example: supplier_code = 101)
SELECT I.item_name
FROM Item I
WHERE I.item_code NOT IN (
  SELECT item_code FROM Can_supply WHERE supplier_code = 101
);
