-- Use the existing library database
USE library;

-- Make sure BOOK and BOOK_AUTHORS exist (optional safety)
CREATE TABLE IF NOT EXISTS BOOK (
  Book_ISBN VARCHAR(20) PRIMARY KEY,
  Title VARCHAR(100) NOT NULL,
  Publisher_Name VARCHAR(50),
  Price DECIMAL(10,2) CHECK (Price > 0),
  Date_Of_Publication DATE,
  Book_Copy INT
);



CREATE TABLE IF NOT EXISTS BOOK_AUTHORS (
  Book_ISBN VARCHAR(20),
  Author_Name VARCHAR(50),
  Author_City VARCHAR(50),
  PRIMARY KEY (Book_ISBN, Author_Name),
  FOREIGN KEY (Book_ISBN) REFERENCES BOOK(Book_ISBN)
);

-- a) INNER JOIN: Book_ISBN, Title, Author_Name
SELECT B.Book_ISBN, B.Title, A.Author_Name
FROM BOOK B
INNER JOIN BOOK_AUTHORS A
ON B.Book_ISBN = A.Book_ISBN;



-- b) LEFT OUTER JOIN: Show all books, even those without authors
SELECT B.Book_ISBN, B.Title, B.Publisher_Name, A.Author_Name
FROM BOOK B
LEFT JOIN BOOK_AUTHORS A
ON B.Book_ISBN = A.Book_ISBN;



-- c) RIGHT OUTER JOIN: Show all authors, even those whose book info may be missing
SELECT B.Book_ISBN, B.Title, B.Publisher_Name, A.Author_Name
FROM BOOK B
RIGHT JOIN BOOK_AUTHORS A
ON B.Book_ISBN = A.Book_ISBN;



-- d) Books whose authors live in 'Pune'
SELECT B.Book_ISBN, B.Title
FROM BOOK B
JOIN BOOK_AUTHORS A
ON B.Book_ISBN = A.Book_ISBN
WHERE A.Author_City = 'Pune';



-- e) Books written by more than 2 authors
SELECT B.Book_ISBN, B.Title
FROM BOOK B
JOIN BOOK_AUTHORS A
ON B.Book_ISBN = A.Book_ISBN
GROUP BY B.Book_ISBN, B.Title
HAVING COUNT(A.Author_Name) > 2;
