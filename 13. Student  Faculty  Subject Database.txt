-- Create and use the database
CREATE DATABASE IF NOT EXISTS college_system;
USE college_system;

-- 1️⃣ Create tables
CREATE TABLE Faculty (
  faculty_code INT PRIMARY KEY,
  faculty_name VARCHAR(100),
  specialization VARCHAR(100)
);

CREATE TABLE Subject (
  subject_code INT PRIMARY KEY,
  subject_name VARCHAR(100),
  faculty_code INT,
  FOREIGN KEY (faculty_code) REFERENCES Faculty(faculty_code)
);

CREATE TABLE Student (
  roll_no INT PRIMARY KEY,
  name VARCHAR(100),
  subject_name VARCHAR(100),
  subject_opted VARCHAR(10)
);

-- 4️⃣ Find number of students enrolled for subject “DBMS”
SELECT COUNT(*) AS Total_Students_DBMS
FROM Student
WHERE subject_name = 'DBMS';

-- 5️⃣ Faculty members who have not offered any subject
SELECT faculty_name
FROM Faculty
WHERE faculty_code NOT IN (SELECT faculty_code FROM Subject);

-- 6️⃣ Subjects not offered by any faculty member
SELECT subject_name
FROM Subject
WHERE faculty_code IS NULL
   OR faculty_code NOT IN (SELECT faculty_code FROM Faculty);
