Step 1: Create Tables and Insert Sample Data
-- 1. Create Course table
CREATE TABLE Course (
    Course_id INT PRIMARY KEY,
    Name VARCHAR2(50),
    Fee NUMBER(10,2),
    Duration VARCHAR2(20),
    Status VARCHAR2(20)
);

-- 2. Create Student table
CREATE TABLE Student (
    Roll_no INT PRIMARY KEY,
    Name VARCHAR2(50),
    Date_of_Birth DATE,
    Course_id INT,
    CONSTRAINT fk_course FOREIGN KEY (Course_id) REFERENCES Course(Course_id)
);

Step 2: Create PL/SQL Procedure
CREATE OR REPLACE PROCEDURE update_course_status IS
BEGIN
    FOR rec IN (
        SELECT c.Course_id, COUNT(s.Roll_no) AS student_count
        FROM Course c
        LEFT JOIN Student s ON c.Course_id = s.Course_id
        GROUP BY c.Course_id
    ) LOOP
        IF rec.student_count < 5 THEN
            UPDATE Course
            SET Status = 'not offered'
            WHERE Course_id = rec.Course_id;
        END IF;
    END LOOP;

    COMMIT;
END;
/

Step 3: Execute the Procedure
BEGIN
    update_course_status;
END;
