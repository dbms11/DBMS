// ==========================================
// a) Count all records from ORDERS
// ==========================================
print("Total Number of Orders:");
db.ORDERS.countDocuments();

// ==========================================
// b) Sum the 'price' field from ORDERS
// ==========================================
print("Total Sum of Price from ORDERS:");
db.ORDERS.aggregate([
  { $group: { _id: null, total_price: { $sum: "$price" } } }
]);

// ==========================================
// c) For each unique cust_id, sum the 'price' field
// ==========================================
print("Sum of Price per Customer:");
db.ORDERS.aggregate([
  { $group: { _id: "$cust_id", total_price: { $sum: "$price" } } }
]);

// ==========================================
// d) For each unique cust_id, sum price, results sorted by sum
// ==========================================
print("Sum of Price per Customer (Sorted by Sum):");
db.ORDERS.aggregate([
  { $group: { _id: "$cust_id", total_price: { $sum: "$price" } } },
  { $sort: { total_price: 1 } }  // ascending order
]);

// ==========================================
// e) For each unique cust_id and ord_date grouping, sum price
// ==========================================
print("Sum of Price Grouped by cust_id and ord_date:");
db.ORDERS.aggregate([
  {
    $group: {
      _id: { cust_id: "$cust_id", ord_date: "$ord_date" },
      total_price: { $sum: "$price" }
    }
  },
  { $sort: { "_id.cust_id": 1, "_id.ord_date": 1 } }
]);
