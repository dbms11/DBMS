a) Count the number of female (F) and male (M) respondents
// Map function
var mapGender = function() {
    emit(this.gender, 1); // key = gender, value = 1
};

// Reduce function
var reduceGender = function(key, values) {
    return Array.sum(values); // sum all counts
};

// Execute map-reduce
db.orders.mapReduce(
    mapGender,
    reduceGender,
    { out: "gender_count" } // output collection
)

// View results
db.gender_count.find()


✅ This will give an output like:

{ "_id" : "F", "value" : 15 }
{ "_id" : "M", "value" : 10 }
