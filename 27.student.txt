-- 1. Create database and use it
CREATE DATABASE IF NOT EXISTS schoolDB;
USE schoolDB;

-- 2. Create STUDENT table
DROP TABLE IF EXISTS STUDENT;
CREATE TABLE STUDENT (
    FIRST_NAME VARCHAR(50),
    MIDDLE_NAME VARCHAR(50),
    LAST_NAME VARCHAR(50),
    STUDENT_ENRLNO INT PRIMARY KEY,
    DATE_OF_BIRTH DATE,
    CLASS INT,
    SECTION CHAR(1),
    GENDER CHAR(1),
    YEAR_OF_JOIN INT,
    ADMISSION_NO INT,
    ADDRESS1 VARCHAR(100),
    ADDRESS2 VARCHAR(100),
    CITY VARCHAR(50),
    STATE VARCHAR(50),
    RESPHONE VARCHAR(15),
    PIN_CODE VARCHAR(10)
);

-- 4. (a) List students in class 6, section A
SELECT 'Query (a): Students in Class 6, Section A' AS Description;
SELECT * FROM STUDENT
WHERE CLASS = 6 AND SECTION = 'A';

-- 5. (b) List students whose first name starts with "A"
SELECT 'Query (b): Students whose first name starts with A' AS Description;
SELECT * FROM STUDENT
WHERE FIRST_NAME LIKE 'A%';

-- 6. (c) List all girl students
SELECT 'Query (c): All girl students' AS Description;
SELECT * FROM STUDENT
WHERE GENDER = 'F';

-- 7. (d) List students whose YEAR_OF_JOIN is 2000
SELECT 'Query (d): Students who joined in 2000' AS Description;
SELECT * FROM STUDENT
WHERE YEAR_OF_JOIN = 2000;

-- 8. (e) Sort students by ADMISSION_NO in ascending order
SELECT 'Query (e): Students sorted by ADMISSION_NO' AS Description;
SELECT * FROM STUDENT
ORDER BY ADMISSION_NO ASC;
