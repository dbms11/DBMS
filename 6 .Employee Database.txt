-- Create and use database
CREATE DATABASE IF NOT EXISTS employee_db;
USE employee_db;


-- Create EmployeeInfo table
CREATE TABLE EmployeeInfo (
  EmpID INT PRIMARY KEY,
  EmpFname VARCHAR(50),
  EmpLname VARCHAR(50),
  Department VARCHAR(30),
  Project VARCHAR(50),
  Address VARCHAR(100),
  DOB DATE,
  Gender CHAR(1)
);



-- Create EmployeePosition table
CREATE TABLE EmployeePosition (
  EmpID INT,
  EmpPosition VARCHAR(50),
  DateOfJoining DATE,
  Salary DECIMAL(10,2),
  PRIMARY KEY (EmpID),
  FOREIGN KEY (EmpID) REFERENCES EmployeeInfo(EmpID)
);

-- i) Fetch EmpFname in UPPERCASE with alias EmpName
SELECT UPPER(EmpFname) AS EmpName
FROM EmployeeInfo;



-- ii) Number of employees working in 'HR' department
SELECT COUNT(*) AS HR_Employees
FROM EmployeeInfo
WHERE Department = 'HR';



-- iii) Employees whose salary is between 50000 and 100000
SELECT EI.EmpFname, EI.EmpLname, EP.Salary
FROM EmployeeInfo EI
JOIN EmployeePosition EP ON EI.EmpID = EP.EmpID
WHERE EP.Salary BETWEEN 50000 AND 100000;



-- iv) Names of employees whose first name begins with 'S'
SELECT EmpFname, EmpLname
FROM EmployeeInfo
WHERE EmpFname LIKE 'S%';



-- v) Fetch top N records (example: top 3 highest salaries)
SELECT EI.EmpFname, EI.EmpLname, EP.Salary
FROM EmployeeInfo EI
JOIN EmployeePosition EP ON EI.EmpID = EP.EmpID
ORDER BY EP.Salary DESC
LIMIT 3;
